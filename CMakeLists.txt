cmake_minimum_required(VERSION 3.16)
project(FlowLens VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FLOWLENS_BUILD_GUI "Build the Qt desktop application" ON)
option(FLOWLENS_SANITIZE "Enable AddressSanitizer for the core verification build" OFF)
if(FLOWLENS_SANITIZE AND MSVC)
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
add_library(flowlens_core src/packet.cpp src/checksum.cpp src/ip_reassembly.cpp src/stream.cpp src/filter.cpp src/pcap_file.cpp src/pcapng_file.cpp)
target_include_directories(flowlens_core PUBLIC include)
if(MSVC)
  target_compile_options(flowlens_core PUBLIC /utf-8 /W4 /permissive-)
else()
  target_compile_options(flowlens_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_compile_definitions(flowlens_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(flowlens_core PUBLIC ws2_32)
  add_executable(flowlens_native_probe tests/native_probe.cpp src/capture.cpp src/process.cpp)
  target_link_libraries(flowlens_native_probe PRIVATE flowlens_core iphlpapi)
endif()
add_executable(flowlens_tests tests/core_tests.cpp)
target_link_libraries(flowlens_tests PRIVATE flowlens_core)
add_executable(flowlens_cli src/cli.cpp)
target_link_libraries(flowlens_cli PRIVATE flowlens_core)
enable_testing()
add_test(NAME core_contracts COMMAND flowlens_tests "${CMAKE_CURRENT_SOURCE_DIR}/samples")
add_executable(flowlens_plus_tests tests/plus_tests.cpp)
target_link_libraries(flowlens_plus_tests PRIVATE flowlens_core)
add_test(NAME plus_contracts COMMAND flowlens_plus_tests "${CMAKE_CURRENT_SOURCE_DIR}/samples")
add_executable(flowlens_attribution_tests tests/attribution_tests.cpp)
target_link_libraries(flowlens_attribution_tests PRIVATE flowlens_core)
add_test(NAME attribution_contracts COMMAND flowlens_attribution_tests "${CMAKE_CURRENT_SOURCE_DIR}/samples")
if(FLOWLENS_BUILD_GUI)
  find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
  add_executable(FlowLens WIN32 src/main.cpp src/main_window.cpp src/capture.cpp src/process.cpp)
  target_link_libraries(FlowLens PRIVATE flowlens_core Qt${QT_VERSION_MAJOR}::Widgets iphlpapi)
  target_compile_definitions(FlowLens PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
